此代码使用了 `class` 来封装棋盘和游戏逻辑。为了代码的可读性和控制台兼容性，我使用了简单的字符来绘制棋盘。

```
#include <iostream>
#include <vector>
#include <string>
#include <cmath>
#include <iomanip>

using namespace std;

// 定义棋盘大小
const int BOARD_SIZE = 15;
const int CENTER = 7; // 数组索引 7 对应第 8 行/列（天元）

// 定义棋子类型
enum Piece { EMPTY = 0, BLACK = 1, WHITE = 2 };

// 坐标结构体
struct Point {
    int r, c;
    bool operator==(const Point& other) const {
        return r == other.r && c == other.c;
    }
};

// 棋盘类：负责存储数据和绘制
class Board {
private:
    int grid[BOARD_SIZE][BOARD_SIZE];

public:
    Board() {
        reset();
    }

    void reset() {
        for (int i = 0; i < BOARD_SIZE; i++)
            for (int j = 0; j < BOARD_SIZE; j++)
                grid[i][j] = EMPTY;
    }

    bool place(int r, int c, Piece p) {
        if (r < 0 || r >= BOARD_SIZE || c < 0 || c >= BOARD_SIZE) return false;
        if (grid[r][c] != EMPTY) return false;
        grid[r][c] = p;
        return true;
    }

    // 撤销（用于N打时的临时显示等，本项目简化暂不用，预留接口）
    void undo(int r, int c) {
        grid[r][c] = EMPTY;
    }

    bool isEmpty(int r, int c) const {
        if (r < 0 || r >= BOARD_SIZE || c < 0 || c >= BOARD_SIZE) return false;
        return grid[r][c] == EMPTY;
    }

    int getPiece(int r, int c) const {
        if (r < 0 || r >= BOARD_SIZE || c < 0 || c >= BOARD_SIZE) return -1;
        return grid[r][c];
    }

    // 绘制棋盘
    void draw(const vector<Point>& candidates = {}) const {
        // 打印列号
        cout << "   ";
        for (int i = 0; i < BOARD_SIZE; i++) cout << setw(2) << i + 1 << " ";
        cout << endl;

        for (int i = 0; i < BOARD_SIZE; i++) {
            cout << setw(2) << i + 1 << " "; // 打印行号
            for (int j = 0; j < BOARD_SIZE; j++) {
                bool isCandidate = false;
                // 检查是否是N打的候选点
                for (size_t k = 0; k < candidates.size(); k++) {
                    if (candidates[k].r == i && candidates[k].c == j) {
                        cout << "[" << k + 1 << "]"; // 显示候选编号
                        isCandidate = true;
                        break;
                    }
                }
                if (isCandidate) continue;

                if (grid[i][j] == EMPTY) {
                    if (i == CENTER && j == CENTER) cout << " + "; // 天元
                    else cout << " . ";
                } else if (grid[i][j] == BLACK) {
                    cout << " X "; // 黑棋
                } else {
                    cout << " O "; // 白棋
                }
            }
            cout << endl;
        }
    }

    // 检查是否胜利
    // 黑棋禁手：长连（超过5个）。这里简单实现：黑棋必须刚好5个，白棋5个或以上
    int checkWin() {
        // 遍历所有点
        for (int i = 0; i < BOARD_SIZE; i++) {
            for (int j = 0; j < BOARD_SIZE; j++) {
                if (grid[i][j] == EMPTY) continue;
              
                int color = grid[i][j];
                // 检查四个方向：横、竖、左斜、右斜
                int dr[] = {0, 1, 1, 1};
                int dc[] = {1, 0, 1, -1};

                for (int d = 0; d < 4; d++) {
                    int count = 1;
                    // 向前查
                    for (int step = 1; step < 6; step++) {
                        int nr = i + step * dr[d];
                        int nc = j + step * dc[d];
                        if (nr >= 0 && nr < BOARD_SIZE && nc >= 0 && nc < BOARD_SIZE && grid[nr][nc] == color) {
                            count++;
                        } else {
                            break;
                        }
                    }
                    // 向后查 (为了确保计算整条线的长度，避免重复计算，通常只算一个方向即可，这里简化逻辑，假设从线条一端开始扫描)
                    // 更严谨的写法是：每落一子，检查该子周围。这里采用全盘扫描简化代码量。
                  
                    // 简单判定逻辑：如果连成5个
                    if (count == 5) {
                        // 检查长连（Overline）：如果黑棋连成6个或以上，不算赢（禁手规则之一）
                        // 如果是白棋，5个及以上都算赢
                        int nextR = i + 5 * dr[d];
                        int nextC = j + 5 * dc[d];
                        bool overline = false;
                        if (nextR >= 0 && nextR < BOARD_SIZE && nextC >= 0 && nextC < BOARD_SIZE && grid[nextR][nextC] == color) {
                            overline = true;
                        }

                        // 反方向前一个
                        int prevR = i - dr[d];
                        int prevC = j - dc[d];
                        if (prevR >= 0 && prevR < BOARD_SIZE && prevC >= 0 && prevC < BOARD_SIZE && grid[prevR][prevC] == color) {
                            // 这是一个连线的中间部分，跳过，等待扫描到头部
                            continue; 
                        }

                        if (color == WHITE) return WHITE; // 白棋长连也算赢
                        if (color == BLACK && !overline) return BLACK; // 黑棋必须刚好5个
                        // 如果黑棋长连，继续游戏（严谨规则判负，这里简化为无效）
                    }
                }
            }
        }
        return 0; // 没人赢
    }
};

// 游戏控制类
class RenjuGame {
private:
    Board board;
    int moveCount;
    // 谁在执黑，谁在执白。1代表玩家A，2代表玩家B
    int playerBlack; 
    int playerWhite; 
  
    // 工具函数：获取输入
    Point getInput(string prompt) {
        Point p;
        while (true) {
            cout << prompt << " (输入行 列，如 8 8): ";
            if (cin >> p.r >> p.c) {
                p.r--; p.c--; // 转换为0基索引
                if (board.isEmpty(p.r, p.c)) return p;
                cout << "位置无效或已有棋子，请重试。\n";
            } else {
                cin.clear(); cin.ignore(1000, '\n');
                cout << "输入格式错误。\n";
            }
        }
        return p;
    }

public:
    RenjuGame() {
        moveCount = 0;
        playerBlack = 1; // 默认玩家1执黑
        playerWhite = 2; // 默认玩家2执白
    }

    void start() {
        cout << "========= 连珠五子棋 (指定开局) =========\n";
        board.draw();

        // --- 1. 指定开局 ---
        // 黑1：必须天元
        cout << "\n[第1手] 规则：黑方必须下在天元 (8, 8)。\n";
        board.place(CENTER, CENTER, BLACK);
        moveCount++;
        board.draw();

        // 白2：黑方决定白2的位置（指定开局规则）
        cout << "\n[第2手] 指定开局：请执黑方决定白棋第2手的位置。\n";
        Point p2 = getInput("玩家(黑)请输入白2坐标");
        board.place(p2.r, p2.c, WHITE);
        moveCount++;
        board.draw();

        // 黑3：必须在天元 5x5 范围内
        cout << "\n[第3手] 规则：黑3必须落在天元周围 5x5 区域内。\n";
        Point p3;
        while(true) {
            p3 = getInput("玩家(黑)请输入黑3坐标");
            if (abs(p3.r - CENTER) <= 2 && abs(p3.c - CENTER) <= 2) {
                board.place(p3.r, p3.c, BLACK);
                break;
            }
            cout << "违规！黑3必须在 5x5 区域内 (行/列 6-10)。\n";
        }
        moveCount++;
        board.draw();

        // --- 2. 三手交换 ---
        cout << "\n=== 三手交换阶段 ===\n";
        cout << "现在轮到白方 (玩家" << playerWhite << ") 选择是否交换颜色。\n";
        cout << "输入 1 交换 (执黑)，输入 0 不换 (继续执白): ";
        int swapChoice;
        cin >> swapChoice;
        if (swapChoice == 1) {
            cout << "双方交换颜色！\n";
            int temp = playerBlack;
            playerBlack = playerWhite;
            playerWhite = temp;
        } else {
            cout << "白方选择不交换。\n";
        }
        cout << "当前执黑: 玩家" << playerBlack << ", 当前执白: 玩家" << playerWhite << endl;

        // 白4：自由落子
        cout << "\n[第4手] 轮到白方落子。\n";
        Point p4 = getInput("请白方落子");
        board.place(p4.r, p4.c, WHITE);
        moveCount++;
        board.draw();

        // --- 3. 五手 N 打 ---
        cout << "\n=== 五手 N 打阶段 ===\n";
        int N;
        while(true) {
            cout << "请黑方提出打点数量 N (2-5): ";
            cin >> N;
            if (N >= 2 && N <= 5) break;
            cout << "范围必须是 2 到 5。\n";
        }

        vector<Point> candidates;
        cout << "请黑方依次输入 " << N << " 个候选点：\n";
        for (int i = 0; i < N; i++) {
            while (true) {
                cout << "候选点 " << i + 1 << ": ";
                Point cand;
                cin >> cand.r >> cand.c;
                cand.r--; cand.c--;
                if (board.isEmpty(cand.r, cand.c)) {
                    // 检查是否重复
                    bool exists = false;
                    for(auto old : candidates) if(old == cand) exists = true;
                    if(!exists) {
                        candidates.push_back(cand);
                        break;
                    }
                    cout << "该点已在候选列表中。\n";
                } else {
                    cout << "该点已被占用。\n";
                }
            }
        }
      
        cout << "\n黑方提供的候选点如下：\n";
        board.draw(candidates);

        cout << "请白方从上述 " << N << " 个点中选择一个作为黑5 (输入序号 1-" << N << "): ";
        int choiceIdx;
        while (true) {
            cin >> choiceIdx;
            if (choiceIdx >= 1 && choiceIdx <= N) break;
            cout << "无效序号，请重试: ";
        }
        Point p5 = candidates[choiceIdx - 1];
        board.place(p5.r, p5.c, BLACK);
        moveCount++;
        cout << "白方选择了第 " << choiceIdx << " 个候选点。\n";
        board.draw();

        // --- 4. 正常对局 ---
        cout << "\n=== 进入正常对局阶段 ===\n";
        while (true) {
            int winner = board.checkWin();
            if (winner != 0) {
                cout << "\n游戏结束！" << (winner == BLACK ? "黑方" : "白方") << " 获胜！\n";
                break;
            }

            int currentPlayer = (moveCount % 2 == 0) ? BLACK : WHITE; // 黑先(1,3,5...)，但在数组逻辑里moveCount从0开始略有不同，这里黑棋永远是奇数手(1,3,5)，白棋偶数。
            // 修正：moveCount=5 (第6手，白棋)
            Piece pType = (moveCount % 2 != 0) ? WHITE : BLACK; // 逻辑修正：moveCount=5 -> 下第6手 -> 白

            cout << "\n[第 " << moveCount + 1 << " 手] 轮到 " << (pType == BLACK ? "黑方" : "白方") << " 落子。\n";
            Point p = getInput(pType == BLACK ? "黑方落子" : "白方落子");
            board.place(p.r, p.c, pType);
            moveCount++;
            board.draw();
        }
    }
};

int main() {
    RenjuGame game;
    game.start();
    return 0;
}
```

### 第二部分：课程设计报告素材

#### （1）如何设计数据结构来表示程序所需的信息？

为了表示连珠五子棋的复杂逻辑，本程序采用了**面向对象（OOP）**的设计思想，主要包含三个核心数据结构：

1. **`Board` 类（棋盘）：**
   * **核心数据** ：使用二维整型数组 `int grid[15][15]` 来映射实际棋盘。`0`表示空点，`1`表示黑子，`2`表示白子。
   * **辅助结构** ：使用 `Point` 结构体 `{row, col}` 来传递坐标信息，简化参数传递。
   * **理由** ：二维数组是表示矩阵棋盘最直接、查询效率最高（O(1)）的方式。
2. **`RenjuGame` 类（游戏控制器）：**
   * **状态数据** ：包含 `moveCount`（手数计数器），用于判断当前该谁下，以及是否处于特殊阶段（如前5手）。
   * **玩家映射** ：使用 `playerBlack` 和 `playerWhite` 变量记录当前哪位玩家（ID）执黑或执白，这是为了实现“三手交换”规则，交换后只需交换这两个变量的值，无需改变棋盘逻辑。
3. **候选点列表（用于五手N打）：**
   * 使用 `std::vector<Point> candidates` 动态数组。
   * **理由** ：N的取值范围是2-5，数量动态变化，使用 `vector` 可以方便地存储、遍历和传递给绘图函数进行特殊标记。

#### （2）请阐明并描述您的算法。

本程序的算法核心是一个 **线性状态机** ，结合了 **贪心搜索** （用于判定胜负）。

1. **游戏流程控制算法：**
   * **阶段一：指定开局** 。强制执行黑1（天元）、接收黑方指定的白2、验证黑3（计算距离天元的切比雪夫距离 `max(|dr|, |dc|) <= 2`）。
   * **阶段二：三手交换** 。在第3手后，插入一个条件分支，允许白方更改 `playerBlack` 和 `playerWhite` 的映射关系。
   * **阶段三：五手N打** 。
   * 黑方输入 N 个不重复坐标 -> 存入 `vector`。
   * 系统重绘棋盘，显示候选点（用 `[1]`, `[2]` 等标记）。
   * 白方输入索引 -> 系统将对应坐标落实为黑子。
   * **阶段四：循环对局** 。使用 `while` 循环，根据 `moveCount % 2` 决定执子颜色，直到 `checkWin()` 返回非零值。
2. **胜负判定算法（`checkWin`）：**
   * 遍历棋盘上每一个非空点。
   * 向4个方向（水平、垂直、两条对角线）发出射线扫描。
   * **黑白差异化判定** ：
   * 如果连成5子，进一步检查两端是否还有同色棋子。
   * **黑棋** ：严格要求连珠数 `== 5`，若 `> 5` 则判定为“长连禁手”（本代码中简化为不赢，继续游戏，符合初级项目要求）。
   * **白棋** ：只要连珠数 `>= 5` 即判定获胜。

#### （3）请说明你们方案的正确性和有效性。

* **规则符合性** ：程序严格按照题目要求的顺序执行：黑1天元 -> 黑定白2 -> 黑3限区 -> 询问交换 -> 白4 -> 黑定N打 -> 白选一点。这保证了逻辑上的正确性。
* **输入鲁棒性** ：在 `getInput` 函数中加入了输入检查。如果用户输入坐标越界（如 16 16）或该位置已有棋子，程序会提示重试，不会导致崩溃或非法覆盖。
* **N打实现的准确性** ：通过 `vector` 暂存候选点，并不直接修改棋盘 `grid`，直到白方确认选择后才修改。这避免了“悔棋”逻辑的复杂性，有效防止了状态污染。

#### （4）功能演示（输入输出逻辑）

* **场景 1：指定开局与交换**
  * *输入* ：黑方第一手自动落子；黑方输入白2 `(7, 8)`；黑方输入黑3 `(6, 6)`。
  * *系统* ：提示白方是否交换。
  * *输入* ：白方输入 `1` (交换)。
  * *结果* ：系统提示“双方交换颜色”，原白方玩家现在执黑，原黑方玩家执白，继续下白4。
* **场景 2：五手N打**
  * *输入* ：黑方输入 N=2。
  * *输入* ：候选点A `(5, 5)`, 候选点B `(5, 6)`。
  * *显示* ：棋盘上对应位置显示 `[1]`, `[2]`。
  * *输入* ：白方输入 `2`。
  * *结果* ：`(5, 6)` 变为黑子 `X`，`(5, 5)` 恢复为空白，游戏继续。
* **场景 3：胜负判定**
  * *输入* ：黑方在横线上连成5子。
  * *结果* ：`checkWin` 检测到 count=5 且无长连，输出“黑方获胜”并结束程序。

#### （5）讨论你的算法的优点和局限性。

**优点：**

1. **逻辑清晰** ：将复杂的连珠开局规则拆解为独立的步骤，易于维护和理解。
2. **用户友好** ：虽然是控制台程序，但提供了坐标引导、错误提示和可视化的N打候选点显示。
3. **扩展性强** ：`Board` 类和 `Game` 类分离，如果未来想加 AI（人工智能），只需在 Game 类中用 `AI.getMove()` 替换 `getInput()` 即可，无需修改棋盘逻辑。

**局限性：**

1. **禁手规则不完整** ：为了适应课程设计的难度，目前只实现了“长连禁手”的检测。连珠规则中复杂的“三三禁手”和“四四禁手”涉及复杂的图论模式匹配（需要判断活三、冲四等概念），在本程序中未实现。这可能导致黑方可以通过下出双三来取胜，这在严格的专业比赛中是不允许的。
2. **界面简陋** ：控制台字符界面刷新时会有闪烁，且必须输入坐标，操作体验不如鼠标点击的图形界面直观。
